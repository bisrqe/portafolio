rules_version = '2';
// Paste into Firebase Console → Storage → Rules, then Publish.
service firebase.storage {
  match /b/{bucket}/o {

    // Admin = the `admin` custom claim (scripts/set-admin-claim.cjs)
    // or the owner's verified Google account
    function isAdmin() {
      return request.auth != null && (
        request.auth.token.admin == true ||
        (request.auth.token.email == 'bismarck@bisrqe.com' && request.auth.token.email_verified == true)
      );
    }

    // CV (PDF) — public read, admin-only write
    match /portfolio_pdfs/{allPaths=**} {
      allow read: if true;
      allow write, delete: if isAdmin()
        && (request.resource == null || request.resource.contentType == 'application/pdf');
    }
  }
}
