rules_version = '2';
// Paste into Firebase Console → Firestore Database → Rules, then Publish.
service cloud.firestore {
  match /databases/{database}/documents {

    // Admin = the `admin` custom claim (scripts/set-admin-claim.cjs)
    // or the owner's verified Google account
    function isAdmin() {
      return request.auth != null && (
        request.auth.token.admin == true ||
        (request.auth.token.email == 'bismarck@bisrqe.com' && request.auth.token.email_verified == true)
      );
    }

    // Public portfolio content: anyone can read, only the admin can write
    match /home/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /projects/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /leadership/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Site-wide settings (visible tag filters, tag translations)
    match /settings/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Legacy photo collection (the Photography page was replaced by Timeless).
    // Readable and writable only by the admin so existing data is not lost.
    match /photos/{docId} {
      allow read, write: if isAdmin();
    }
  }
}
